    guidellm.collectors

    Collectors gather data from outside GuideLLM while each benchmark runs.

    Configure them with --collector kind=.... Each collector reports one result per benchmark, stored in that benchmark's collectors list in the order the collectors were configured.

    Collector

    Bases: RegistryMixin[type['Collector']], ABC

    Gather data from outside GuideLLM for each benchmark of a run.

    A collector is reused across the benchmarks of a run. For each benchmark, :meth:start is called before its requests begin, :meth:stop after they end, and :meth:result once the measurement window is known. Every collector returns a result for every benchmark, even an empty one, so results line up with the order the collectors were configured in.

    class Collector(RegistryMixin[type["Collector"]], ABC):
    """
    Gather data from outside GuideLLM for each benchmark of a run.

    A collector is reused across the benchmarks of a run. For each benchmark,
    :meth:`start` is called before its requests begin, :meth:`stop` after they
    end, and :meth:`result` once the measurement window is known. Every collector
    returns a result for every benchmark, even an empty one, so results line up
    with the order the collectors were configured in.
    """

    def __init__(self, kind: str) -> None:
        """
        :param kind: The kind this collector was configured with
        """
        self.kind = kind

    @classmethod
    @abstractmethod
    def from_args(cls, args: CollectorArgs) -> Collector:
        """
        Create a collector from its arguments.

        :param args: Arguments for this collector's kind
        :return: The configured collector
        """
        ...

    @classmethod
    def resolve(cls, args: CollectorArgs) -> Collector:
        """
        Resolve collector arguments into a collector.

        :param args: Collector arguments with a registered kind
        :return: The configured collector
        :raises ValueError: If the kind is not registered
        """
        collector_class = cls.get_registered_object(args.kind)
        if collector_class is None:
            available = list(cls.registry.keys()) if cls.registry else []
            raise ValueError(
                f"Collector kind '{args.kind}' is not registered. "
                f"Available kinds: {available}"
            )

        return collector_class.from_args(args)

    @abstractmethod
    async def start(self) -> None:
        """Begin collecting for a new benchmark."""
        ...

    @abstractmethod
    async def stop(self) -> None:
        """Finish collecting once the benchmark's requests have ended."""
        ...

    @abstractmethod
    def result(self, start_time: float, end_time: float) -> dict[str, Any]:
        """
        Return the result for the latest benchmark.

        :param start_time: Start of the measurement window in seconds since epoch
        :param end_time: End of the measurement window in seconds since epoch
        :return: JSON-serializable result, which may be empty
        """
        ...

    async def close(self) -> None:  # noqa: B027
        """Release any resources held across benchmarks."""

    class CollectorProcess:
    """
    Own the child process that runs a benchmark's collectors.

    Example:
        ::

            collectors = CollectorProcess(args)
            await collectors.start()
            try:
                await collectors.benchmark_started()
                ...  # run the benchmark
                await collectors.benchmark_stopped()
                results = await collectors.results(start_time, end_time)
            finally:
                await collectors.shutdown()
    """

    def __init__(self, args: list[CollectorArgs]):
        """
        :param args: Arguments for each collector, in the order results are returned
        """
        self.args = args
        self._process: BaseProcess | None = None
        self._connection: Connection | None = None

    async def start(self) -> None:
        """
        Start the child process and create the collectors in it.

        :raises RuntimeError: If a collector cannot be created
        """
        context = get_context(settings.mp_context_type)
        parent_connection, child_connection = context.Pipe()
        self._process = context.Process(
            target=_run_collectors,
            args=(child_connection, [arg.model_dump() for arg in self.args]),
            name="guidellm-collectors",
            daemon=True,
        )
        self._process.start()
        child_connection.close()
        self._connection = parent_connection
        await self._request("ready")

    async def benchmark_started(self) -> None:
        """Start every collector for a new benchmark."""
        await self._request("start")

    async def benchmark_stopped(self) -> None:
        """Stop every collector once the benchmark's requests have ended."""
        await self._request("stop")

    async def results(self, start_time: float, end_time: float) -> list[dict[str, Any]]:
        """
        Return each collector's result for the latest benchmark.

        :param start_time: Start of the measurement window in seconds since epoch
        :param end_time: End of the measurement window in seconds since epoch
        :return: One result per collector, in configuration order
        """
        return await self._request("result", start_time, end_time)

    async def shutdown(self) -> None:
        """Close the collectors and wait for the child process to exit."""
        if self._process is None or self._connection is None:
            return
        try:
            await self._request("close")
        except (EOFError, OSError, RuntimeError) as err:
            logger.warning("Collector process did not close cleanly: {}", err)
        self._connection.close()
        await asyncio.to_thread(self._process.join, _SHUTDOWN_TIMEOUT)
        if self._process.is_alive():
            self._process.terminate()
        self._process = None
        self._connection = None

    async def _request(self, command: str, *params: Any) -> Any:
        if self._connection is None:
            raise RuntimeError("Collector process has not been started")
        if command != "ready":
            self._connection.send((command, *params))
        status, payload = await asyncio.to_thread(self._connection.recv)
        if status != "ok":
            raise RuntimeError(f"Collector process failed on {command!r}: {payload}")

        return payload

    class OneshotCollector(Collector, ABC):
    """
    Collect once per benchmark, after its requests have ended.

    Suited to data that does not change during a benchmark, such as a server's
    configuration. It is still collected for every benchmark, since a run's
    benchmarks may be configured differently.
    """

    def __init__(self, kind: str) -> None:
        """
        :param kind: The kind this collector was configured with
        """
        super().__init__(kind)
        self._result: dict[str, Any] = {}

    @abstractmethod
    async def collect(self) -> dict[str, Any]:
        """
        Collect this benchmark's data.

        :return: JSON-serializable result, which may be empty
        """
        ...

    async def start(self) -> None:
        """Discard the previous benchmark's result."""
        self._result = {}

    async def stop(self) -> None:
        """Collect the result for the benchmark that just ended."""
        self._result = await self.collect()

    def result(self, start_time: float, end_time: float) -> dict[str, Any]:  # noqa: ARG002
        """
        :param start_time: Start of the measurement window in seconds since epoch
        :param end_time: End of the measurement window in seconds since epoch
        :return: The result collected when the benchmark ended
        """
        return self._result

    class PeriodicCollector(Collector, ABC):
    """
    Sample at a fixed interval while a benchmark runs.

    Samples once when the benchmark starts, every ``interval`` seconds while it
    runs, and once more when it stops, so the measurement window is covered at
    both ends. Subclasses summarize the samples in :meth:`result`.
    """

    def __init__(self, kind: str, interval: float) -> None:
        """
        :param kind: The kind this collector was configured with
        :param interval: Seconds between samples while a benchmark runs
        """
        super().__init__(kind)
        self.interval = interval
        self._task: asyncio.Task[None] | None = None

    @abstractmethod
    async def reset(self) -> None:
        """Discard the previous benchmark's samples before a new one starts."""
        ...

    @abstractmethod
    async def sample(self) -> None:
        """Take one sample. Failures should be recorded, not raised."""
        ...

    async def start(self) -> None:
        """Reset, sample once, then sample every interval."""
        await self.reset()
        await self.sample()
        self._task = asyncio.create_task(self._sample_periodically())

    async def stop(self) -> None:
        """Stop the periodic samples and take a final sample."""
        if self._task is not None:
            self._task.cancel()
            with contextlib.suppress(asyncio.CancelledError):
                await self._task
            self._task = None
        await self.sample()

    async def _sample_periodically(self) -> None:
        while True:
            await asyncio.sleep(self.interval)
            await self.sample()

class PrometheusCollector(PeriodicCollector):
    """
    Scrape a Prometheus text-format endpoint, such as vLLM's ``/metrics``.

    Scrape failures are counted and logged, and never interrupt the benchmark.

    Example:
        ::

            collector = PrometheusCollector(
                url="http://localhost:8000/metrics",
                interval=1.0,
                timeout=5.0,
                metrics=["vllm:num_requests_running"],
            )
            await collector.start()
            ...
            await collector.stop()
            result = collector.result(start_time, end_time)
    """

    def __init__(
        self,
        url: str,
        interval: float,
        timeout: float,
        metrics: list[str],
        transport: httpx2.AsyncBaseTransport | None = None,
        kind: str = "prometheus",
    ):
        """
        :param url: URL of the Prometheus metrics endpoint
        :param interval: Seconds between scrapes while a benchmark runs
        :param timeout: Seconds to wait for a single scrape
        :param metrics: Metric family names to collect
        :param transport: Optional httpx2 transport, used to substitute the server
        :param kind: The kind this collector was configured with
        """
        super().__init__(kind, interval)
        self.url = url
        self.timeout = timeout
        self.metrics = metrics
        self._transport = transport
        self._client: httpx2.AsyncClient | None = None
        self._snapshots: list[PrometheusSnapshot] = []
        self._scrape_errors = 0

    @classmethod
    def from_args(cls, args: CollectorArgs) -> PrometheusCollector:
        """
        Create a collector from Prometheus collector arguments.

        :param args: Prometheus collector arguments
        :return: The configured collector
        :raises TypeError: If the arguments are not Prometheus arguments
        """
        if not isinstance(args, PrometheusCollectorArgs):
            raise TypeError(
                f"Expected PrometheusCollectorArgs, got {type(args).__name__}"
            )

        return cls(
            url=args.url,
            interval=args.interval,
            timeout=args.timeout,
            metrics=args.metrics,
            kind=args.kind,
        )

    async def reset(self) -> None:
        """Discard the previous benchmark's scrapes."""
        self._snapshots = []
        self._scrape_errors = 0

    async def sample(self) -> None:
        """Scrape the endpoint once, counting a failure instead of raising."""
        if self._client is None:
            self._client = httpx2.AsyncClient(
                timeout=self.timeout, transport=self._transport
            )
        requested_at = time.time()
        try:
            response = await self._client.get(
                self.url, headers={"Accept": _ACCEPT_HEADER}
            )
            response.raise_for_status()
        except (httpx2.HTTPError, httpx2.InvalidURL) as err:
            self._scrape_errors += 1
            if self._scrape_errors == 1:
                logger.warning("Failed to scrape metrics from {}: {}", self.url, err)
            return
        # The midpoint of the request is the best estimate of when it was sampled.
        timestamp = (requested_at + time.time()) / 2
        self._snapshots.append(
            parse_prometheus_text(response.text, self.metrics, timestamp)
        )

    def result(self, start_time: float, end_time: float) -> dict[str, Any]:
        """
        Summarize the latest benchmark's scrapes over its measurement window.

        See :func:`summarize_snapshots` for how each metric type is summarized.

        :param start_time: Start of the measurement window in seconds since epoch
        :param end_time: End of the measurement window in seconds since epoch
        :return: The :class:`PrometheusSummary` as a JSON-serializable dict
        """
        if end_time - start_time < 2 * self.interval:
            logger.warning(
                "Measurement window of {:.2f}s is shorter than twice the {}s "
                "scrape interval of {}, so its metrics are approximate",
                end_time - start_time,
                self.interval,
                self.url,
            )

        return summarize_snapshots(
            self.url, self._snapshots, self._scrape_errors, start_time, end_time
        ).model_dump(mode="json")

    async def close(self) -> None:
        """Close the HTTP client."""
        if self._client is not None:
            await self._client.aclose()
            self._client = None

    def histogram_quantile(quantile: float, buckets: dict[float, float]) -> float | None:
    """
    Estimate a quantile from cumulative histogram bucket counts.

    Follows Prometheus ``histogram_quantile``: the quantile is interpolated
    linearly within the bucket that contains it, and falls back to the highest
    finite bound when it lands in the ``+Inf`` bucket.

    :param quantile: Quantile to estimate, between 0 and 1
    :param buckets: Cumulative counts by upper bound, including ``+Inf``
    :return: The estimated value, or None when the histogram is empty
    """
    bounds = sorted(buckets)
    if not bounds or not math.isinf(bounds[-1]):
        return None
    counts: list[float] = []
    for bound in bounds:
        counts.append(max(buckets[bound], counts[-1] if counts else 0.0))
    total = counts[-1]
    if total <= 0:
        return None

    rank = quantile * total
    index = bisect_left(counts, rank)
    if math.isinf(bounds[index]):
        return bounds[index - 1] if index > 0 else None

    lower_bound = bounds[index - 1] if index > 0 else min(0.0, bounds[0])
    lower_count = counts[index - 1] if index > 0 else 0.0
    in_bucket = counts[index] - lower_count
    if in_bucket <= 0:
        return bounds[index]

    return (
        lower_bound + (bounds[index] - lower_bound) * (rank - lower_count) / in_bucket
    )

    def parse_prometheus_text(
    text: str, families: list[str], timestamp: float
) -> PrometheusSnapshot:
    """
    Parse the configured metric families from Prometheus text exposition format.

    Only lines that start with a configured family name are parsed, so large
    responses with many unrelated metrics stay cheap. A counter family matches
    both ``name`` and ``name_total`` as declared by its ``# TYPE`` line. Families
    without a ``# TYPE`` line, types other than gauge, counter and histogram,
    malformed lines, and non-finite values are skipped.

    :param text: Response body in Prometheus text exposition format
    :param families: Metric family names to collect, without counter suffixes
    :param timestamp: Time of the scrape in seconds since epoch
    :return: Snapshot of the configured families found in the text
    """
    snapshot = PrometheusSnapshot(timestamp=timestamp)
    prefixes = tuple(families)
    families_by_type_name = {name: name for name in families}
    families_by_type_name.update({f"{name}_total": name for name in families})
    samples: dict[str, tuple[str, str, str]] = {}

    for raw_line in text.splitlines():
        line = raw_line.strip()
        if line.startswith("# TYPE "):
            parts = line.split()
            if len(parts) < 4 or parts[2] not in families_by_type_name:  # noqa: PLR2004
                continue
            family, metric_type = families_by_type_name[parts[2]], parts[3]
            if metric_type in ("gauge", "counter", "histogram"):
                for sample_name, part in _family_samples(family, metric_type).items():
                    samples[sample_name] = (family, metric_type, part)
            continue
        if not line.startswith(prefixes):
            continue

        try:
            name, labels, value = _split_sample(line)
        except (ValueError, IndexError):
            continue
        if name in samples and math.isfinite(value):
            _record_sample(snapshot, *samples[name], labels, value)

    return snapshot
